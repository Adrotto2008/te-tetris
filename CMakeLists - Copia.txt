cmake_minimum_required(VERSION 3.20)

project(te_tetris VERSION 0.1.0 LANGUAGES CXX)

# ============================================================
# Standard C++
# ============================================================

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(TE_TETRIS_FETCH_DEPS "Scarica le dipendenze mancanti con FetchContent" ON)
option(TE_TETRIS_ENABLE_TLS "Abilita TLS in IXWebSocket" OFF)

set(TE_TETRIS_APP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/versions/current")
set(TE_TETRIS_MAIN "${TE_TETRIS_APP_ROOT}/main.cpp")
set(TE_TETRIS_SRC_DIR "${TE_TETRIS_APP_ROOT}/src")
set(TE_TETRIS_INCLUDE_DIR "${TE_TETRIS_APP_ROOT}/include")

if(NOT EXISTS "${TE_TETRIS_MAIN}")
    message(FATAL_ERROR "main.cpp non trovato in ${TE_TETRIS_MAIN}")
endif()

if(NOT EXISTS "${TE_TETRIS_INCLUDE_DIR}")
    message(FATAL_ERROR "Cartella include non trovata: ${TE_TETRIS_INCLUDE_DIR}")
endif()

if(TE_TETRIS_FETCH_DEPS)
    include(FetchContent)
endif()

# ============================================================
# Dipendenze: ZLIB
# ============================================================

find_package(ZLIB QUIET)

if(NOT TARGET ZLIB::ZLIB AND TE_TETRIS_FETCH_DEPS)
    FetchContent_Declare(
        zlib
        GIT_REPOSITORY https://github.com/madler/zlib.git
        GIT_TAG v1.3.1
    )

    set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)

    FetchContent_MakeAvailable(zlib)

    if(TARGET zlibstatic)
        add_library(ZLIB::ZLIB ALIAS zlibstatic)
    elseif(TARGET zlib)
        add_library(ZLIB::ZLIB ALIAS zlib)
    endif()
endif()

if(NOT TARGET ZLIB::ZLIB)
    message(FATAL_ERROR "ZLIB non trovata. Installa ZLIB o abilita TE_TETRIS_FETCH_DEPS.")
endif()

# ============================================================
# Dipendenze: nlohmann_json
# ============================================================

find_package(nlohmann_json CONFIG QUIET)

if(NOT TARGET nlohmann_json::nlohmann_json AND TE_TETRIS_FETCH_DEPS)
    FetchContent_Declare(
        nlohmann_json
        GIT_REPOSITORY https://github.com/nlohmann/json.git
        GIT_TAG v3.11.3
    )
    FetchContent_MakeAvailable(nlohmann_json)
endif()

if(NOT TARGET nlohmann_json::nlohmann_json)
    message(FATAL_ERROR "nlohmann_json non trovato. Installa il package o abilita TE_TETRIS_FETCH_DEPS.")
endif()

# ============================================================
# Dipendenze: cpp-httplib (header-only)
# ============================================================

set(TE_TETRIS_HTTPLIB_TARGET "")
find_package(httplib CONFIG QUIET)

if(TARGET httplib::httplib)
    set(TE_TETRIS_HTTPLIB_TARGET httplib::httplib)
elseif(TE_TETRIS_FETCH_DEPS)
    FetchContent_Declare(
        cpp_httplib
        GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git
        GIT_TAG v0.15.3
    )
    FetchContent_MakeAvailable(cpp_httplib)

    if(TARGET httplib::httplib)
        set(TE_TETRIS_HTTPLIB_TARGET httplib::httplib)
    else()
        add_library(te_tetris_httplib INTERFACE)
        target_include_directories(te_tetris_httplib INTERFACE "${cpp_httplib_SOURCE_DIR}")
        set(TE_TETRIS_HTTPLIB_TARGET te_tetris_httplib)
    endif()
endif()

if(TE_TETRIS_HTTPLIB_TARGET STREQUAL "")
    message(FATAL_ERROR "cpp-httplib non trovato. Installa il package o abilita TE_TETRIS_FETCH_DEPS.")
endif()

# ============================================================
# Dipendenze: IXWebSocket
# ============================================================

find_package(ixwebsocket CONFIG QUIET)

if(NOT TARGET ixwebsocket AND TE_TETRIS_FETCH_DEPS)
    set(IXWEBSOCKET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(IXWEBSOCKET_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(IXWEBSOCKET_BUILD_WITH_TLS ${TE_TETRIS_ENABLE_TLS} CACHE BOOL "" FORCE)
    set(IXWEBSOCKET_INSTALL OFF CACHE BOOL "" FORCE)
    set(USE_ZLIB OFF CACHE BOOL "" FORCE)

    FetchContent_Declare(
        ixwebsocket
        GIT_REPOSITORY https://github.com/machinezone/IXWebSocket.git
        GIT_TAG v11.4.5
    )
    FetchContent_MakeAvailable(ixwebsocket)
endif()

if(NOT TARGET ixwebsocket)
    message(FATAL_ERROR "IXWebSocket non trovato. Installa il package o abilita TE_TETRIS_FETCH_DEPS.")
endif()

# ============================================================
# Dipendenze: SFML
# ============================================================

find_package(SFML 2.6 COMPONENTS system window graphics audio QUIET)
if(NOT SFML_FOUND AND TE_TETRIS_FETCH_DEPS)
    set(SFML_BUILD_AUDIO ON CACHE BOOL "" FORCE)
    set(SFML_BUILD_GRAPHICS ON CACHE BOOL "" FORCE)
    set(SFML_BUILD_WINDOW ON CACHE BOOL "" FORCE)
    set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_DOC OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)

    FetchContent_Declare(
        sfml
        GIT_REPOSITORY https://github.com/SFML/SFML.git
        GIT_TAG 2.6.1
    )
    FetchContent_MakeAvailable(sfml)
endif()

if(NOT SFML_FOUND AND NOT TARGET sfml-system AND NOT TARGET SFML::System)
    find_package(SFML 2.5 COMPONENTS system window graphics audio REQUIRED)
endif()

set(TE_TETRIS_SFML_LIBS)
if(TARGET SFML::System)
    list(APPEND TE_TETRIS_SFML_LIBS SFML::System SFML::Window SFML::Graphics SFML::Audio)
else()
    list(APPEND TE_TETRIS_SFML_LIBS sfml-system sfml-window sfml-graphics sfml-audio)
endif()

# ============================================================
# Sorgenti progetto
# ============================================================

file(GLOB_RECURSE TE_TETRIS_SRC_FILES CONFIGURE_DEPENDS
    "${TE_TETRIS_SRC_DIR}/*.cpp"
)

add_executable(te_tetris)

target_sources(te_tetris PRIVATE
    "${TE_TETRIS_MAIN}"
    ${TE_TETRIS_SRC_FILES}
)

target_include_directories(te_tetris PRIVATE
    "${TE_TETRIS_INCLUDE_DIR}"
)

# ============================================================
# Link librerie
# ============================================================

target_link_libraries(te_tetris PRIVATE
    ${TE_TETRIS_SFML_LIBS}
    ixwebsocket
    nlohmann_json::nlohmann_json
    ZLIB::ZLIB
    ${TE_TETRIS_HTTPLIB_TARGET}
)

if(UNIX AND NOT APPLE)
    target_link_libraries(te_tetris PRIVATE
        pthread
        dl
        m
        rt
        udev
    )
endif()

# ============================================================
# Output
# ============================================================

set_target_properties(te_tetris PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin"
)

# ============================================================
# Messaggi
# ============================================================

message(STATUS "======================================")
message(STATUS " te_tetris configurato")
message(STATUS " C++${CMAKE_CXX_STANDARD}")
message(STATUS " root codice  -> ${TE_TETRIS_APP_ROOT}")
message(STATUS " WebSocket    -> ixwebsocket")
message(STATUS " json         -> nlohmann_json")
message(STATUS " http         -> cpp-httplib")
message(STATUS " zlib         -> ZLIB::ZLIB")
message(STATUS " SFML         -> system window graphics audio")
message(STATUS " Output       -> ${CMAKE_CURRENT_SOURCE_DIR}/bin")
message(STATUS "======================================")
